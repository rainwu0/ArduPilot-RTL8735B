# RTL8735B 的 SDK 端建置：把 waf 編好的 ArduPilot 靜態庫與 Realtek SDK 的原始碼、預編譯庫
# 連結成 application.ntz，再依 SDK 的後處理流程產生可燒錄的 flash_ntz.bin。
# 由 Tools/ardupilotwaf/rtl8735b.py 呼叫。
#
# 傳入的變數：
#   AMEBAPRO2_SDK      SDK 根目錄
#   ARDUPILOT_CMD      copter、plane、rover、sub 或 none
#   WAF_BUILD_TARGET   waf 的建置目標（examples/xxx 時連結範例程式）
#   ARDUPILOT_LIB      waf 放靜態庫的目錄（lib/）
#   ARDUPILOT_BIN      waf 放載具靜態庫的目錄（lib/bin/）
#   CMAKE_TOOLCHAIN_FILE  SDK 的 toolchain.cmake（旗標、-wrap 與 -D 都來自它）
#   RTL8735B_WIFI_CREDENTIALS_SOURCE  waf 在建置目錄產生的 wifi_credentials.c（hwdef/scripts/rtl8735b_wifi_config.py），
#                      未設定 Wi-Fi 時是 SSID 長度 0 的版本；內容可能含帳密，只在建置目錄
#   RTL8735B_APP_DIR   板子的應用程式目錄（hwdef 的 env RTL8735B_APP_DIR，絕對路徑）；空字串表示沒有
#   RTL8735B_PARTITION_TABLE  waf 在建置目錄產生的分割表 JSON（hwdef 的 FLASH_LAYOUT，
#                      hwdef/scripts/rtl8735b_flash_layout.py），取代 SDK 範例專案的那一份
#
# SDK 的 config.cmake 會在 configure 時以 sed -i 改寫 SDK 內的檔案，但預設值下內容不變；
# 檔案時間由 sdk_mtime.cmake 在內容未變時還原，避免 SDK 端整份重編。
# 另外會在 SDK 的 inc/ 產生 build_info.h（SDK 自己以 .gitignore 排除）。

cmake_minimum_required(VERSION 3.18)

project(ardupilot_rtl8735b C CXX ASM)

if(NOT DEFINED AMEBAPRO2_SDK)
    message(FATAL_ERROR "AMEBAPRO2_SDK is not set (path to the Realtek RTL8735B SDK)")
endif()
set(sdk_root "${AMEBAPRO2_SDK}")
set(prj_root "${sdk_root}/project/realtek_amebapro2_v0_example")
set(app application.ntz)

include(${CMAKE_CURRENT_LIST_DIR}/sdk_mtime.cmake)
rtl8735b_sdk_mtime_save(${CMAKE_CURRENT_BINARY_DIR}/sdk_sed_saved)
include(${prj_root}/GCC-RELEASE/config.cmake)
rtl8735b_sdk_mtime_restore(${CMAKE_CURRENT_BINARY_DIR}/sdk_sed_saved)
# 分割表依 hwdef 的 FLASH_LAYOUT：SDK 的 config.cmake 無條件設定 POSTBUILD_PART，所以在 include 之後改。
# 下面取出的 SDK 打包步驟把它複製成建置目錄的 amebapro2_partitiontable.json 再轉成分割表
if(NOT RTL8735B_PARTITION_TABLE OR NOT EXISTS "${RTL8735B_PARTITION_TABLE}")
    message(FATAL_ERROR "RTL8735B_PARTITION_TABLE is not set or missing (generated by Tools/ardupilotwaf/rtl8735b.py)")
endif()
set(POSTBUILD_PART "${RTL8735B_PARTITION_TABLE}")
include(${prj_root}/GCC-RELEASE/includepath.cmake)
# 韌體映像格式要求附上 ISP IQ 與 VOE 映像；沿用 SDK 的產生方式
include(${prj_root}/GCC-RELEASE/sensor_iq.cmake)

# 引導程式：與 SDK 範例相同，從 SDK 的 bootloader 專案產生 boot.bin 與 boot_fcs.bin
add_subdirectory(${prj_root}/GCC-RELEASE/bootloader bootloader)

##############################################################
# SDK 原始碼：建置時由 SDK 的 application.cmake 取得（sdk_import.cmake）。
# TODO(rtl8735b): SDK 的預編譯庫參考 Wi-Fi、lwIP 與影像模組的符號，所以這些原始碼全部帶進來
# 以確保能連結；之後依連結結果裁掉不需要的部分。
##############################################################
include(${CMAKE_CURRENT_LIST_DIR}/sdk_import.cmake)
set(freertos_dir ${sdk_root}/component/os/freertos/${freertos}/Source)
rtl8735b_sdk_sources(sdk_sources)

# 與 SDK 範例的 app_flags 相同
set(sdk_defines
    CONFIG_BUILD_RAM=1
    CONFIG_PLATFORM_8735B
    CONFIG_RTL8735B_PLATFORM=1
    CONFIG_SYSTEM_TIME64=1
)

# include 路徑：SDK 的 includepath.cmake 加上範例 application.cmake 另列的目錄（去掉 NN 與範例目錄）
set(sdk_includes
    "${sdk_root}/component/soc/8735b/fwlib/rtl8735b/lib/include/pub"
    "${sdk_root}/component/soc/8735b/cmsis/rtl8735b/lib/include/pub"
    ${inc_path}
    ${freertos_dir}/portable/GCC/ARM_CM33/non_secure
    ${freertos_dir}/portable/GCC/ARM_CM33/secure
    ${sdk_root}/component/soc/8735b/fwlib/rtl8735b/lib/source/ram/video/voe_bin
    ${sdk_root}/component/soc/8735b/fwlib/rtl8735b/lib/source/ram/video/osd
    ${sdk_root}/component/soc/8735b/misc/platform
    ${sdk_root}/component/video/driver/RTL8735B
    ${sdk_root}/component/video/osd2
    ${sdk_root}/component/media/mmfv2
    ${sdk_root}/component/media/rtp_codec
    ${sdk_root}/component/mbed/hal_ext
    ${sdk_root}/component/file_system/ftl
    ${sdk_root}/component/file_system/system_data
    ${sdk_root}/component/file_system/fwfs
    ${sdk_root}/component/wifi/wpa_supplicant/src
    ${sdk_root}/component/wifi/driver/src/core/option
    ${sdk_root}/component/wifi/wifi_config
    ${sdk_root}/component/ssl/ssl_ram_map/rom
    ${sdk_root}/component/media/framework
    ${prj_root}/src
)

add_library(sdk_objs OBJECT ${sdk_sources})
target_compile_definitions(sdk_objs PRIVATE ${sdk_defines})

# malloc 的包裝：toolchain.cmake 以 -Wl,-wrap,malloc 把 malloc 導向 __wrap_malloc。ArduPilot 要求
# malloc 回傳清零的記憶體（AP_Common/c++.cpp 的 __wrap_malloc 經 __real_malloc 配置後 memset），
# SDK 的 libc_wrap.c 也定義 __wrap_malloc 但只呼叫 pvPortMalloc、不清零。這裡把 SDK 的那個
# 改名讓開，ArduPilot 的版本經 newlib malloc → _malloc_r（同樣被 -wrap 導向 SDK 的
# __wrap__malloc_r → pvPortMalloc）取得記憶體。
set_source_files_properties(
    ${sdk_root}/component/soc/8735b/misc/utilities/source/ram/libc_wrap.c
    PROPERTIES COMPILE_DEFINITIONS "__wrap_malloc=rtl8735b_sdk_wrap_malloc")
target_compile_options(sdk_objs PRIVATE $<$<COMPILE_LANGUAGE:C>:${OUTSRC_WARN_ERR_FLAGS}>)
target_include_directories(sdk_objs PUBLIC ${sdk_includes})

##############################################################
# 韌體：入口 main.c 加 SDK 物件，連結 ArduPilot 靜態庫與 SDK 預編譯庫
##############################################################
if(NOT DEFINED RTL8735B_WIFI_CREDENTIALS_SOURCE OR NOT EXISTS "${RTL8735B_WIFI_CREDENTIALS_SOURCE}")
    message(FATAL_ERROR "RTL8735B_WIFI_CREDENTIALS_SOURCE is not set or missing (generated by Tools/ardupilotwaf/rtl8735b.py)")
endif()
add_executable(${app} main.c ${RTL8735B_WIFI_CREDENTIALS_SOURCE} $<TARGET_OBJECTS:sdk_objs>)
target_compile_definitions(${app} PRIVATE ${sdk_defines})
target_compile_options(${app} PRIVATE $<$<COMPILE_LANGUAGE:C>:${WARN_ERR_FLAGS}>)
target_include_directories(${app} PUBLIC ${sdk_includes})

# 板子的應用程式（選用）：目錄中的 .c 與入口一起編譯，沿用 SDK 的 include 與巨集，
# 由 main.c 呼叫其中的 rtl8735b_app_init()
if(RTL8735B_APP_DIR)
    file(GLOB app_sources CONFIGURE_DEPENDS "${RTL8735B_APP_DIR}/*.c")
    if(NOT app_sources)
        message(FATAL_ERROR "RTL8735B_APP_DIR has no .c files: ${RTL8735B_APP_DIR}")
    endif()
    message(STATUS "RTL8735B_APP_DIR=${RTL8735B_APP_DIR}")
    target_sources(${app} PRIVATE ${app_sources})
    target_include_directories(${app} PRIVATE ${RTL8735B_APP_DIR})
endif()

if(NOT DEFINED ARDUPILOT_CMD)
    set(ARDUPILOT_CMD "none")
endif()
message(STATUS "ARDUPILOT_CMD=${ARDUPILOT_CMD}")
message(STATUS "WAF_BUILD_TARGET=${WAF_BUILD_TARGET}")

string(REGEX MATCH "^(examples|tool)/" is_example "${WAF_BUILD_TARGET}")
if(is_example)
    string(REPLACE "/" ";" parts ${WAF_BUILD_TARGET})
    list(GET parts 0 example_base)
    list(GET parts 1 example_name)
    set(ardupilot_libs
        "${ARDUPILOT_LIB}/${example_base}/lib${example_name}.a"
        "${ARDUPILOT_LIB}/libap.a")
elseif(ARDUPILOT_CMD STREQUAL "copter")
    set(ardupilot_libs "${ARDUPILOT_BIN}/libarducopter.a" "${ARDUPILOT_LIB}/libArduCopter_libs.a")
elseif(ARDUPILOT_CMD STREQUAL "plane")
    set(ardupilot_libs "${ARDUPILOT_BIN}/libarduplane.a" "${ARDUPILOT_LIB}/libArduPlane_libs.a")
elseif(ARDUPILOT_CMD STREQUAL "rover")
    set(ardupilot_libs "${ARDUPILOT_BIN}/libardurover.a" "${ARDUPILOT_LIB}/libRover_libs.a")
elseif(ARDUPILOT_CMD STREQUAL "sub")
    set(ardupilot_libs "${ARDUPILOT_BIN}/libardusub.a" "${ARDUPILOT_LIB}/libArduSub_libs.a")
else()
    set(ardupilot_libs "")
endif()

add_library(hal_pmc_lib STATIC IMPORTED)
set_property(TARGET hal_pmc_lib PROPERTY IMPORTED_LOCATION
    ${sdk_root}/component/soc/8735b/fwlib/rtl8735b/lib/lib/hal_pmc.a)

target_link_directories(${app} PRIVATE ${prj_root}/GCC-RELEASE/application/output)
target_link_libraries(${app}
    hal_pmc_lib
    -Wl,--start-group
    ${ardupilot_libs}
    bsp_ntz
    wlan
    wps
    video_ntz
    mmf
    nn
    lightsensor
    ispfeature
    soc_ntz
    -Wl,--end-group
    stdc++
    m
    c
    gcc
    nosys
)

# 連結腳本開頭 INCLUDE "romsym_is.so"（ROM 符號表），放在 SDK 專案的 ROM/GCC/
set(ld_script ${prj_root}/GCC-RELEASE/application/rtl8735b_ram.ld)
target_link_options(${app} PUBLIC
    "LINKER:SHELL:-L ${prj_root}/GCC-RELEASE/ROM/GCC"
    "LINKER:SHELL:-L ${CMAKE_CURRENT_BINARY_DIR}"
    "LINKER:SHELL:-T ${ld_script}"
    "LINKER:SHELL:-Map=${CMAKE_CURRENT_BINARY_DIR}/${app}.map"
)
set_target_properties(${app} PROPERTIES LINK_DEPENDS ${ld_script})
add_dependencies(${app} fcs_isp_iq)

##############################################################
# 打包：ELF 後處理、韌體轉換、金鑰與憑證、分割表、燒錄映像組合，
# 建置時由 SDK 的 cmake 檔取得（sdk_import.cmake），不在本庫重列。
##############################################################
if(NOT MPCHIP)
    message(FATAL_ERROR "只支援 SDK 的 MPCHIP 組態")
endif()
set(postbuild_cfg ${POSTBUILD_FW_NTZ})
set(firmware_name firmware_ntz.bin)
set(boot_image bootloader/boot.bin)
set(fcs_param bootloader/boot_fcs.bin)
set(boot_dep bootloader)
set(fw firmware_ntz.bin)
set(isp_iq firmware_isp_iq.bin)
set(symbols ${app}.symbols)
set(bttrace APP.trace)
set(app_dep ${app})
set(target flash_ntz)
# ELF 後處理（符號表、trace 區段）；不另外產生 .asm 檔，也不複製到 output/
rtl8735b_sdk_extract(sdk_code ${prj_root}/GCC-RELEASE/application/application.cmake
    --block "add_custom_command(" TARGET "\${app}" POST_BUILD
    --drop OBJEDUMP --drop PLAT_COPY --drop "[ \t]output")
cmake_language(EVAL CODE "${sdk_code}")
# 韌體轉換（firmware_ntz.bin）
rtl8735b_sdk_extract(sdk_code ${prj_root}/GCC-RELEASE/application/CMakeLists.txt
    --block "add_custom_command(" TARGET "\${app}" POST_BUILD --require "FIRMWARE ..firmware_name")
cmake_language(EVAL CODE "${sdk_code}")
# 金鑰、憑證與分割表，組合成 flash_ntz.bin
foreach(tokens IN ITEMS "add_custom_command(;OUTPUT;key_public.json"
                        "add_custom_target(;plain_img"
                        "add_custom_target(;flash")
    rtl8735b_sdk_extract(sdk_code ${prj_root}/GCC-RELEASE/CMakeLists.txt
        --within "if(MPCHIP)" "if(UNITEST)" --block ${tokens})
    cmake_language(EVAL CODE "${sdk_code}")
endforeach()

##############################################################
# showinc：把 include 目錄與巨集寫成清單，給 waf 編譯 ArduPilot 時用
##############################################################
add_custom_target(showinc ALL
    COMMAND echo "$<REMOVE_DUPLICATES:$<TARGET_PROPERTY:${app},INCLUDE_DIRECTORIES>>" > includes.list
    COMMAND echo "$<REMOVE_DUPLICATES:$<TARGET_PROPERTY:${app},COMPILE_DEFINITIONS>>" > defines.list
    VERBATIM
    BYPRODUCTS includes.list defines.list
    COMMAND_EXPAND_LISTS
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
)
